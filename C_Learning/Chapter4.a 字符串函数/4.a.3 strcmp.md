比较两个字符串s1, s2
s1 == s2 ---> 0
s1 >   s2 -----> >0
s1  <  s2 -----> <0
```c
int mycmp(const char *s1,const char *s2){
	int idx = 0;
	while ( s1[idx] == s2[idx]  && s1[idx]!= '\0')
		idx++;
}
return s1[idx] - s2[idx]
```


```c
#include <stdio.h>

int mycmp(const char *s1, const char *s2){
    while(*s1 == *s2 && *s1 != "\0"){
        s1++;
        s2++;
    }
    return *s2 - *s1;
}

int main(void) {
    char s1 [] ="Hello,World";
    char s2 [] ="Hello,world";
    printf("mycmp:%d",mycmp(s1,s2));

return 0;
}
```
