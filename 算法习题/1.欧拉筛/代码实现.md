#include <stdio.h>  
#include <stdbool.h>  
#include <math.h>  
#include <stdlib.h>  
int cnt = 0;  
void euler_sieve(int num);  
int main(void){  
    int number;  
    printf("Enter a number:\n");  
    scanf("%d",&number);  
    euler_sieve(number);  
    printf("%d,%d primenumber\n",number,cnt);  
}  
void euler_sieve(int num){  
    int* primeNumber = (int*)malloc((num+1)*sizeof(int));  
    bool isPrime[num+1];  
    for(int i=0; i<num+1; i++){  
        isPrime[i] = true;  
    }  
    for(int i=2; i<=num; i++){  
        if (isPrime[i]){  
            primeNumber[cnt++] = i;  
        }  
        for(int j=0; j<cnt && i*primeNumber[j] <= num; j++){  
            isPrime[i*primeNumber[j]] = false;  
            if (i % primeNumber[j] == 0){  
                break;  
            }  
        }  
    }  
}
