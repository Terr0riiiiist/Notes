### 数组个数获取
```c
typedef struct{
	int *array;
	int size;
} Array;

Array array_create(int init_size) {
	Array a;   
	a.array = (int*)malloc(sizeof(int)*init_size);  
	a.size  = init_size;
	return a;
}

int array_size(const Array *a){
	return a->size;
}  //封装，可以保护a->size;
```
---
### 访问索引处的元素
```c
int* array_at(Array *a, int index){
	assert(index > 0 && index < a->size); 
	return &(a->array[index]);
}
```
