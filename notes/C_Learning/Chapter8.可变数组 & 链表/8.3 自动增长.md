```c
void array_inflate(Array *a, int more_size){
	int *p = (int*)malloc(sizeof(int)*(a->size + more_size));
	int i;
	for (i=0; i<a->size; i++){
		p[i] = a->array[i];
	}
	free(a->array);
	a->array = p;
	a->size += more_size;
} //不使用realloc函数
```

```c
int* array_at(Array *a, int index){
	if(index >= a->size){
		array_inflate(a, (index/BLOCK_SIZE+1)*BLOCK_SIZE-a->size);
	}
	return &(a->array[index]);
}	
```

$$\text{more\_size} = (\lfloor \frac{\text{index}}{\text{BLOCK\_SIZE}} \rfloor + 1) \times \text{BLOCK\_SIZE} - \text{a}\rightarrow\text{size}$$
